% IEEEtran asks for Times in T1. Tectonic runs XeTeX, whose default TU encoding
% has no `ptm', so without these two lines every Times shape falls back to
% Computer Modern -- 170 font warnings, and a page count longer than the real one
% because CM is wider. newtx supplies Times-compatible text and math faces.
\usepackage[T1]{fontenc}
\usepackage{amsmath}
% newtxmath must come after amsmath and REPLACES amssymb/amsfonts -- it provides
% the AMS symbol set itself, and loading both makes \Bbbk multiply defined and
% halts the build.
\usepackage{newtxtext,newtxmath}
% IEEEtran itself asks for `ptm' by name in a few places (title, author block).
% newtx installs its faces under `ntxtlf', so those requests fell back to the
% default family and printed five font warnings. Alias ptm onto newtx's Times so
% the build is warning-clean and the title is set in the face the class intends.
\DeclareFontFamily{TU}{ptm}{}
\DeclareFontShape{TU}{ptm}{m}{n}{<->ssub*ntxtlf/m/n}{}
\DeclareFontShape{TU}{ptm}{m}{it}{<->ssub*ntxtlf/m/it}{}
\DeclareFontShape{TU}{ptm}{b}{n}{<->ssub*ntxtlf/b/n}{}
\DeclareFontShape{TU}{ptm}{b}{it}{<->ssub*ntxtlf/b/it}{}
\DeclareFontShape{TU}{ptm}{bx}{n}{<->ssub*ntxtlf/b/n}{}
\DeclareFontShape{TU}{ptm}{bx}{it}{<->ssub*ntxtlf/b/it}{}
